cmake_minimum_required(VERSION 3.25)
project(ee_skintokens LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SKINTOKENS_SOURCE_DIR "" CACHE PATH "Checkout of the pinned SkinTokens C++ runtime")
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${SKINTOKENS_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE upstream_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT upstream_revision STREQUAL "43aac7b0fe45f5cd802365e6964c13bfc5e5fb52")
  message(FATAL_ERROR "SkinTokens source must match the reviewed runtime revision")
endif()
# Portable baseline: statically linked CPU GGML, no runtime backend discovery dependency.
set(SKINTOKENS_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(SKINTOKENS_DYNAMIC_BACKENDS OFF CACHE BOOL "" FORCE)
set(SKINTOKENS_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(SKINTOKENS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${SKINTOKENS_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/skintokens" EXCLUDE_FROM_ALL)
add_library(ee_skintokens SHARED bridge.cpp)
target_link_libraries(ee_skintokens PRIVATE skintokens::skintokens)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(ee_skintokens PRIVATE ${JNI_INCLUDE_DIRS})
endif()
# Package the two shared libraries side by side; CPU GGML is static.
if(APPLE)
  set_target_properties(ee_skintokens PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX AND NOT ANDROID)
  set_target_properties(ee_skintokens PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
if(ANDROID)
  # Android's native loader expects unversioned .so names.
  set_target_properties(skintokens PROPERTIES VERSION "" SOVERSION "")
endif()
install(TARGETS ee_skintokens skintokens
  RUNTIME DESTINATION common LIBRARY DESTINATION common)
